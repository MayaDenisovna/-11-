def valid(f):
    def wrap(*args, **kw):
        for v in args:
            if v <= 0:
                return "Ошибка: аргументы должны быть больше нуля"
        for v in kw.values():
            if v <= 0:
                return "Ошибка: аргументы должны быть больше нуля"
        return f(*args, **kw)
    return wrap

@valid
def pr(a, b):
    return a * b

def memo(f):
    cash = dict()
    def wrap(n):
        if n not in cash:
            cash[n] = f(n)
        return cash[n]
    return wrap

@memo
def fib(n):
    if n <= 1:
        return n
    return fib(n - 1) + fib(n - 2)


def merge(d1, d2):
    res = dict(d1)
    for k, v in d2.items():
        if k in res and isinstance(res[k], dict) and isinstance(v, dict):
            res[k] = merge(res[k], v)
        else:
            res[k] = v
    return res
d1 = {'a': 1, 'b': {'x': 10, 'y': 20}}
d2 = {'b': {'y': 30, 'z': 40}, 'c': 3}
mrg = merge(d1, d2)
print(mrg)

def find(docs, tags):
    t_set = set(tags)
    match = set()
    for d_id, d_tgs in docs.items():
        if t_set.issubset(d_tgs):
            match.add(d_id)
    return match

import time
def prof(f):
    def wrap(*args, **kw):
        t0 = time.perf_counter()
        res = f(*args, **kw)
        t1 = time.perf_counter()
        dt = t1 - t0
        print({'execution_time_sec': dt})
        return res
    return wrap
@prof
def r_sum(n):
    if n <= 1:
        return n
    return n + r_sum(n - 1)
res = r_sum(10)
print(res)
